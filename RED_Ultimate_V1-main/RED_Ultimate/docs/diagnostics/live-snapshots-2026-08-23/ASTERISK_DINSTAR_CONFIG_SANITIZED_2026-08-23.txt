  ; strip +967 / 00967 / 967 / 0
  same => n,ExecIf($["${_NUM:0:4}"="+967"]?Set(_NUM=${_NUM:4}))
  same => n,ExecIf($["${_NUM:0:5}"="00967"]?Set(_NUM=${_NUM:5}))
  same => n,ExecIf($["${_NUM:0:3}"="967"]?Set(_NUM=${_NUM:3}))
  same => n,ExecIf($["${_NUM:0:1}"="0"]?Set(_NUM=${_NUM:1}))
  same => n,GotoIf($["${_NUM:0:2}"="70"]?ytelecom)
  same => n,GotoIf($["${_NUM:0:2}"="71"]?sabafon)
  same => n,GotoIf($["${_NUM:0:2}"="73"]?you)
  same => n,GotoIf($["${_NUM:0:2}"="77"]?yemobile)
  same => n,GotoIf($["${_NUM:0:2}"="78"]?yemobile)
  same => n,Set(CHANNEL(tone_zone)=ye)
  same => n,Return()
  same => n(ytelecom),Set(CHANNEL(tone_zone)=ye-ytelecom)
  same => n,Return()
  same => n(sabafon),Set(CHANNEL(tone_zone)=ye-sabafon)
